class 員工():
    def __init__(self,姓名,月薪):
        self.姓名 = 姓名
        self.月薪 = int(月薪)
        self.時薪 = 月薪/240

    def 工作日(self,時數) -> None:
        self.時數 = 時數

    def 休假日(self,時數):
        self.時數 = 時數

    def 國定假日(self,時數):
        self.時數 = 時數

月薪 = int(input("請輸入您的月薪"))
我的員工 = 員工(姓名,月薪)
